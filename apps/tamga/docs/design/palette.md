```dart
// Primary orange
static const Color primary = Color(0xFFF06A2E);    // 500
static const Color primaryLight = Color(0xFFFFDBC8); // 100
static const Color primaryDark = Color(0xFFCC5824);  // 600
static const Color onPrimary = Color(0xFFFFFFFF);

// Neutrals
static const Color background = Color(0xFFFDF6F3);   // 50

static const Color mutedForeground = Color(0xFF625F5D); // 400
```

```css
:root {

}
```
